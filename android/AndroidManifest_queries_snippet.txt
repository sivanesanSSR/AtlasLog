<!-- Add this inside <manifest> in android/app/src/main/AndroidManifest.xml,
     as a sibling of <application>. Required on Android 11+ (API 30+) so
     url_launcher's canLaunchUrl() can detect WhatsApp and the SMS app;
     without this, canLaunchUrl silently returns false even when the
     apps are installed. -->

<queries>
    <intent>
        <action android:name="android.intent.action.VIEW" />
        <data android:scheme="https" />
    </intent>
    <intent>
        <action android:name="android.intent.action.VIEW" />
        <data android:scheme="whatsapp" />
    </intent>
    <intent>
        <action android:name="android.intent.action.SENDTO" />
        <data android:scheme="sms" />
    </intent>
    <package android:name="com.whatsapp" />
</queries>
